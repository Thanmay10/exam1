require(MASS)

ad=read.csv(url)
dim(ad)
head(ad)
model=glm(admit~gre+gpa+rank,data=ad,family=binomial())
summary(model)
fitted(model)
predicted=ifelse(predict(model,type="response")>0.5,1,0)
print(predicted)
install.packages("caret")
library(caret)
cm=confusionMatrix(data=factor(predicted),reference=factor(ad$admit),positive="1")

17 lines 
data(mtcars)
model_all=lm(mpg~.,data=mtcars)
summary(model_all)
v=cor(mtcars$mpg,mtcars[, -1])
print(v)
model=lm(mpg~drat+qsec+vs+am+gear,data=mtcars)
summary(model)
par(mfrow=c(2,2))
plot(model)
par(mfrw=c(1,1))
pred=predict(model)
plot(mtcars$mpg,pred,main="Actual vs predicted",xlab="actual",ylab="Predicted",col="blue",pch=19)
abline(0,1,col="black",lwd=2)
coefs=coef(model)[-1]
print(coefs)
barplot(coefs,main="Coeffients",ylab="Coe value",col=ifelse(coefs>0,"pink","green"))
abline(h=0,col="black",lwd=2)

  Bed = c(2, 2, 3, 3, 4, 4),
